            @RequestParam(required = false) MultipartFile levelsFile,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        Establishment current = est.get();
        Long tenantId = current.getId();
        try {
            Set<String> names = new LinkedHashSet<>();
            if (levelsFile != null && !levelsFile.isEmpty()) {
                String content = new String(levelsFile.getBytes(), StandardCharsets.UTF_8);
                addLevelTokens(content, names);
            }
            addLevelTokens(levelsInput, names);
            if (names.isEmpty()) {
                redirectAttributes.addFlashAttribute("error", "Aucun niveau fourni.");
                return "redirect:/tenant/app?tenantId=" + tenantId + "#module-classes";
            }
            int created = 0;
            int skipped = 0;
            for (String name : names) {
                if (levelRepository.findByEstablishmentAndNameIgnoreCase(current, name).isPresent()) {
                    skipped++;
                    continue;
                }
                levelRepository.save(new Level(name, current));
                created++;
            }
            if (created > 0) {
                redirectAttributes.addFlashAttribute("success", "Niveaux importes : " + created);
            } else {
                redirectAttributes.addFlashAttribute("info", "Aucun niveau ajoute. Doublons ignores : " + skipped);
            }
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur import niveaux : " + ex.getMessage());
        }
        return "redirect:/tenant/app?tenantId=" + tenantId + "#module-classes";
    }
