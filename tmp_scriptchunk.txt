        function openStudentModal(mode = 'create', btn) {
            if (!studentModal) return;
            resetStudentForm();
            if ((mode === 'edit' || mode === 'view') && btn?.dataset) {
                if (studentIdInput) studentIdInput.value = btn.dataset.id || '';
                if (studentFirstName) studentFirstName.value = btn.dataset.firstName || '';
                if (studentLastName) studentLastName.value = btn.dataset.lastName || '';
                if (studentEmail) studentEmail.value = btn.dataset.email || '';
                if (studentPhoneFixed) studentPhoneFixed.value = btn.dataset.phoneFixed || '';
                if (studentPhoneMobile) studentPhoneMobile.value = btn.dataset.phoneMobile || '';
                if (studentClass) studentClass.value = btn.dataset.classId || '';
                if (studentObservation) studentObservation.value = btn.dataset.observation || '';
            }
            if (mode === 'edit') {
                if (studentTitle) studentTitle.textContent = "Modifier l'apprenant";
                if (studentSubmit) studentSubmit.textContent = 'Mettre à jour';
                studentForm.action = `/tenant/app/students/update?tenantId=${tenantId}`;
            }
            if (mode === 'view') {
                if (studentTitle) studentTitle.textContent = 'Informations apprenant';
                if (studentSubtitle) studentSubtitle.textContent = 'Consultation des informations.';
                studentForm.action = '#';
                studentForm.onsubmit = () => false;
                if (studentSubmit) studentSubmit.style.display = 'none';
                if (studentFirstName) studentFirstName.readOnly = true;
                if (studentLastName) studentLastName.readOnly = true;
                if (studentEmail) {
                    studentEmail.readOnly = true;
                    studentEmail.required = false;
                }
                if (studentPhoneFixed) studentPhoneFixed.readOnly = true;
                if (studentPhoneMobile) studentPhoneMobile.readOnly = true;
                if (studentClass) studentClass.disabled = true;
                if (studentObservation) studentObservation.readOnly = true;
            }
            studentModal.style.display = 'flex';
        }
