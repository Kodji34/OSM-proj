            @RequestParam(required = false) String projectsInput,

        try {
            Set<String> lines = new LinkedHashSet<>();
            if (projectsFile != null && !projectsFile.isEmpty()) {
                String content = new String(projectsFile.getBytes(), StandardCharsets.UTF_8);
                lines.addAll(List.of(content.split("\\r?\\n")));
            }
            if (projectsInput != null && !projectsInput.isBlank()) {
                lines.addAll(List.of(projectsInput.split("\\r?\\n")));
            }
            for (String line : lines) {
                if (line == null) continue;
                String trimmed = line.trim();
                if (trimmed.isBlank()) continue;

                    continue;
                }
                String[] parts = trimmed.split("[,;\\t]");
                if (parts.length < 1) {
                    skipped++;
                    continue;

                    continue;
