    @GetMapping("/tenant/dashboard")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN')")
    public String tenantDashboard(Model model) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) {
            return "error-tenant";
        }
        Establishment current = est.get();
        model.addAttribute("est", current);
        List<User> tenantUsers = userRepository.findTopByEstablishmentIdOrderByIdDesc(current.getId())
                .map(List::of)
                .orElseGet(List::of);
        model.addAttribute("tenantUsers", tenantUsers);
        model.addAttribute("roles", tenantAssignableRoles());

        List<ModuleDefinition> baseModules = moduleDefinitionRepository.findByCategory(ModuleCategory.BASE);
        List<ModuleDefinition> premiumModules = moduleDefinitionRepository.findByCategory(ModuleCategory.PREMIUM);
        Map<Long, ModuleSubscription> subsByModule = moduleSubscriptionRepository
                .findByEstablishmentId(current.getId())
                .stream()
                .collect(Collectors.toMap(s -> s.getModule().getId(), s -> s, (a, b) -> a));
        List<ModuleItem> premiumItems = premiumModules.stream()
                .map(m -> {
                    ModuleSubscription sub = subsByModule.get(m.getId());
                    ModuleStatus status = sub != null ? sub.getStatus() : ModuleStatus.INACTIVE;
                    return new ModuleItem(m, status);
                })
                .toList();

        Map<Long, ModuleStatus> moduleStatusById = subsByModule.values().stream()
                .collect(Collectors.toMap(s -> s.getModule().getId(), ModuleSubscription::getStatus, (a, b) -> a));

        model.addAttribute("baseModules", baseModules);
        model.addAttribute("premiumModules", premiumItems);
        model.addAttribute("moduleStatusById", moduleStatusById);

        List<SupportTicket> tenantTickets = supportTicketRepository.findByEstablishment_IdOrderByUpdatedAtDesc(current.getId());
        model.addAttribute("tenantTickets", tenantTickets);
        if (!tenantTickets.isEmpty()) {
            List<Long> ids = tenantTickets.stream().map(SupportTicket::getId).toList();
            Map<Long, List<SupportTicketMessage>> messagesByTicketId = supportTicketMessageRepository
                    .findByTicket_IdInOrderByCreatedAtAsc(ids)
                    .stream()
                    .collect(Collectors.groupingBy(m -> m.getTicket().getId()));
            model.addAttribute("tenantTicketMessagesById", messagesByTicketId);
        } else {
            model.addAttribute("tenantTicketMessagesById", Map.of());
        }
        return "tenant-dashboard";
    }

    @GetMapping("/tenant/app")
    @PreAuthorize("hasAnyRole('ADMIN','USER','SUPER_ADMIN','DIRECTEUR','SECRETAIRE','COMPTABLE','DIRECTEUR_RH','ENSEIGNANT','APPRENANT')")
    public String tenantApp(Model model) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) {
            return "error-tenant";
        }
        Establishment current = est.get();
        model.addAttribute("est", current);

        List<User> tenantUsers = userRepository.findByEstablishmentId(current.getId());
        long recentLogins = tenantUsers.stream()
                .filter(u -> u.getLastLoginAt() != null && u.getLastLoginAt().isAfter(java.time.LocalDateTime.now().minusHours(24)))
                .count();
        model.addAttribute("tenantUsers", tenantUsers);
        model.addAttribute("recentLogins", recentLogins);
        model.addAttribute("employees", employeeRepository.findByEstablishmentId(current.getId()));
        model.addAttribute("roles", tenantAssignableRoles());
        model.addAttribute("academicYears", academicYearRepository.findByEstablishmentIdOrderByStartDateDesc(current.getId()));
        List<Level> levels = levelRepository.findByEstablishmentIdOrderByNameAsc(current.getId());
        List<SchoolClass> classes = schoolClassRepository.findByEstablishmentIdOrderByNameAsc(current.getId());
        List<Room> rooms = roomRepository.findByEstablishmentIdOrderByNameAsc(current.getId());
        List<Subject> subjects = subjectRepository.findByEstablishmentIdOrderByTitleAsc(current.getId());
        List<PedagogicalProject> projects = pedagogicalProjectRepository.findByEstablishmentIdOrderByIdDesc(current.getId());
        List<ProfessionalInsertion> insertions = professionalInsertionRepository.findByEstablishmentIdOrderByStartDateDesc(current.getId());
        model.addAttribute("levels", levels);
        model.addAttribute("classes", classes);
        model.addAttribute("rooms", rooms);
        model.addAttribute("subjects", subjects);
        model.addAttribute("levelsCount", levels.size());
        model.addAttribute("classesCount", classes.size());
        model.addAttribute("roomsCount", rooms.size());
        model.addAttribute("subjectsCount", subjects.size());
        model.addAttribute("projects", projects);
        model.addAttribute("insertions", insertions);
        model.addAttribute("projectsCount", projects.size());
        model.addAttribute("insertionsCount", insertions.size());
        List<CareerPath> careerPaths = careerPathRepository.findByEstablishmentIdOrderByNameAsc(current.getId());
        model.addAttribute("careerPaths", careerPaths);
        model.addAttribute("careerPathsCount", careerPaths.size());
        List<User> studentUsers = userRepository.findByEstablishmentIdAndRoles_NameIn(current.getId(), List.of("APPRENANT"))
                .stream()
                .sorted((a, b) -> {
                    String lastA = a.getLastName() != null ? a.getLastName() : "";
                    String lastB = b.getLastName() != null ? b.getLastName() : "";
                    int lastCompare = lastA.compareToIgnoreCase(lastB);
                    if (lastCompare != 0) return lastCompare;
                    String firstA = a.getFirstName() != null ? a.getFirstName() : "";
                    String firstB = b.getFirstName() != null ? b.getFirstName() : "";
                    return firstA.compareToIgnoreCase(firstB);
                })
                .toList();
        model.addAttribute("studentUsers", studentUsers);
        model.addAttribute("studentsCount", studentUsers.size());
        List<User> teacherUsers = userRepository.findByEstablishmentIdAndRoles_NameIn(current.getId(), List.of("ENSEIGNANT"))
                .stream()
                .sorted((a, b) -> {
                    String lastA = a.getLastName() != null ? a.getLastName() : "";
                    String lastB = b.getLastName() != null ? b.getLastName() : "";
                    int lastCompare = lastA.compareToIgnoreCase(lastB);
                    if (lastCompare != 0) return lastCompare;
                    String firstA = a.getFirstName() != null ? a.getFirstName() : "";
                    String firstB = b.getFirstName() != null ? b.getFirstName() : "";
                    return firstA.compareToIgnoreCase(firstB);
                })
                .toList();
        long titulairesCount = teacherUsers.stream()
                .filter(u -> u.getTeacherType() != null && "TITULAIRE".equalsIgnoreCase(u.getTeacherType()))
                .count();
        long vacatairesCount = teacherUsers.stream()
                .filter(u -> u.getTeacherType() != null && "VACATAIRE".equalsIgnoreCase(u.getTeacherType()))
                .count();
        model.addAttribute("teacherUsers", teacherUsers);
        model.addAttribute("teachersCount", teacherUsers.size());
        model.addAttribute("teachersTitulaireCount", titulairesCount);
        model.addAttribute("teachersVacataireCount", vacatairesCount);
        List<String> learnerRoles = List.of("APPRENANT");
        Map<Long, Long> classEffectifs = new java.util.HashMap<>();
        for (SchoolClass schoolClass : classes) {
            Long classId = schoolClass.getId();
            long count = 0;
            if (classId != null) {
                count = userRepository.countByEstablishmentIdAndRoles_NameInAndSchoolClass_Id(current.getId(), learnerRoles, classId);
            }
            classEffectifs.put(classId, count);
        }
        model.addAttribute("classEffectifs", classEffectifs);
        model.addAttribute("staffCandidates", buildCandidates(tenantUsers, Set.of(
                "ADMIN", "DIRECTEUR", "SECRETAIRE", "COMPTABLE", "DIRECTEUR_RH", "ENSEIGNANT"
        )));
        model.addAttribute("teacherCandidates", buildCandidates(tenantUsers, Set.of("ENSEIGNANT")));
        model.addAttribute("studentCandidates", buildCandidates(tenantUsers, Set.of("APPRENANT")));

        List<ModuleDefinition> baseModules = moduleDefinitionRepository.findByCategory(ModuleCategory.BASE)
                .stream()
                .filter(m -> !"PROFILE".equalsIgnoreCase(m.getCode())) // profil supprimé côté app client
                .toList();
        List<ModuleDefinition> premiumModules = moduleDefinitionRepository.findByCategory(ModuleCategory.PREMIUM);
        Map<Long, ModuleSubscription> subsByModule = moduleSubscriptionRepository
                .findByEstablishmentId(current.getId())
                .stream()
                .collect(Collectors.toMap(s -> s.getModule().getId(), s -> s, (a, b) -> a));
        List<ModuleItem> premiumItems = premiumModules.stream()
                .map(m -> {
                    ModuleSubscription sub = subsByModule.get(m.getId());
                    ModuleStatus status = sub != null ? sub.getStatus() : ModuleStatus.INACTIVE;
                    return new ModuleItem(m, status);
                })
                .toList();

        model.addAttribute("baseModules", baseModules);
        model.addAttribute("premiumModules", premiumItems);
        return "tenant-app";
    }

    @PostMapping("/tenant/app/academic-year")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN')")
    public String createAcademicYear(
            @RequestParam String academicYear,
            @RequestParam String periodType,
            @RequestParam String startDate,
            @RequestParam String endDate,
            @RequestParam String term1Start,
            @RequestParam String term1End,
            @RequestParam String term2Start,
            @RequestParam String term2End,
            @RequestParam(required = false) String term3Start,
            @RequestParam(required = false) String term3End,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        Long tenantId = est.get().getId();
        try {
            String yearLabel = academicYear != null ? academicYear.trim() : "";
            if (yearLabel.isBlank()) {
                redirectAttributes.addFlashAttribute("error", "L'annee scolaire est requise.");
                return "redirect:/tenant/app?tenantId=" + tenantId + "#module-academic_year";
            }
            if (academicYearRepository.findByAcademicYearAndEstablishmentId(yearLabel, tenantId).isPresent()) {
                redirectAttributes.addFlashAttribute("error", "Cette annee scolaire existe deja.");
                return "redirect:/tenant/app?tenantId=" + tenantId + "#module-academic_year";
            }

            AcademicYear year = new AcademicYear();
            year.setEstablishment(est.get());
            applyAcademicYearData(yearLabel, periodType, startDate, endDate,
                    term1Start, term1End, term2Start, term2End, term3Start, term3End, year);
            academicYearRepository.save(year);
            redirectAttributes.addFlashAttribute("success", "Annee scolaire enregistree.");
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur annee scolaire : " + ex.getMessage());
        }
        return "redirect:/tenant/app?tenantId=" + tenantId + "#module-academic_year";
    }

    @PostMapping("/tenant/app/academic-year/update")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN')")
    public String updateAcademicYear(
            @RequestParam Long yearId,
            @RequestParam String academicYear,
            @RequestParam String periodType,
            @RequestParam String startDate,
            @RequestParam String endDate,
            @RequestParam String term1Start,
            @RequestParam String term1End,
            @RequestParam String term2Start,
            @RequestParam String term2End,
            @RequestParam(required = false) String term3Start,
            @RequestParam(required = false) String term3End,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        Long tenantId = est.get().getId();
        try {
            AcademicYear year = academicYearRepository.findById(yearId)
                    .orElseThrow(() -> new IllegalArgumentException("Annee introuvable."));
            if (!tenantId.equals(year.getEstablishment().getId())) {
                throw new IllegalStateException("Annee hors de ce locataire.");
            }
            String yearLabel = academicYear != null ? academicYear.trim() : "";
            if (yearLabel.isBlank()) {
                throw new IllegalArgumentException("L'annee scolaire est requise.");
            }
            academicYearRepository.findByAcademicYearAndEstablishmentId(yearLabel, tenantId)
                    .filter(existing -> !existing.getId().equals(yearId))
                    .ifPresent(existing -> {
                        throw new IllegalArgumentException("Cette annee scolaire existe deja.");
                    });

            applyAcademicYearData(yearLabel, periodType, startDate, endDate,
                    term1Start, term1End, term2Start, term2End, term3Start, term3End, year);
            academicYearRepository.save(year);
            redirectAttributes.addFlashAttribute("success", "Annee scolaire mise a jour.");
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur annee scolaire : " + ex.getMessage());
        }
        return "redirect:/tenant/app?tenantId=" + tenantId + "#module-academic_year";
    }

    @PostMapping("/tenant/app/levels/import")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN')")
    public String importLevels(
            @RequestParam(required = false) String levelsInput,
            @RequestParam(required = false) MultipartFile levelsFile,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        Establishment current = est.get();
        Long tenantId = current.getId();
        try {
            Set<String> names = new LinkedHashSet<>();
            if (levelsFile != null && !levelsFile.isEmpty()) {
                String content = new String(levelsFile.getBytes(), StandardCharsets.UTF_8);
                addLevelTokens(content, names);
            }
            addLevelTokens(levelsInput, names);
            if (names.isEmpty()) {
                redirectAttributes.addFlashAttribute("error", "Aucun niveau fourni.");
                return "redirect:/tenant/app?tenantId=" + tenantId + "#module-classes";
            }
            int created = 0;
            int skipped = 0;
            for (String name : names) {
                if (levelRepository.findByEstablishmentAndNameIgnoreCase(current, name).isPresent()) {
                    skipped++;
                    continue;
                }
                levelRepository.save(new Level(name, current));
                created++;
            }
            if (created > 0) {
                redirectAttributes.addFlashAttribute("success", "Niveaux importes : " + created);
            } else {
                redirectAttributes.addFlashAttribute("info", "Aucun niveau ajoute. Doublons ignores : " + skipped);
            }
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur import niveaux : " + ex.getMessage());
        }
        return "redirect:/tenant/app?tenantId=" + tenantId + "#module-classes";
    }

    @PostMapping("/tenant/app/classes")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN')")
    public String saveClass(
            @RequestParam(required = false) Long classId,
            @RequestParam String name,
            @RequestParam(required = false) Long levelId,
            @RequestParam(required = false) Long academicYearId,
            @RequestParam(required = false) Long roomId,
            @RequestParam(required = false) String responsibleStaff,
            @RequestParam(required = false) String responsibleTeacher,
            @RequestParam(required = false) String delegateStudent,
            @RequestParam(required = false, defaultValue = "false") boolean acceptRoomConflict,
            @RequestParam(required = false, defaultValue = "false") boolean acceptCapacityOverflow,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        Establishment current = est.get();
        Long tenantId = current.getId();
        try {
            String trimmedName = name != null ? name.trim() : "";
            if (trimmedName.isBlank()) {
                redirectAttributes.addFlashAttribute("error", "Le nom de la classe est requis.");
                return "redirect:/tenant/app?tenantId=" + tenantId + "#module-classes";
            }
            SchoolClass schoolClass = classId != null
                    ? schoolClassRepository.findById(classId)
                        .orElseThrow(() -> new IllegalArgumentException("Classe introuvable."))
                    : new SchoolClass();
            if (schoolClass.getEstablishment() != null && !tenantId.equals(schoolClass.getEstablishment().getId())) {
                throw new IllegalStateException("Classe hors de ce locataire.");
            }
            final AcademicYear year;
            if (academicYearId != null) {
                year = academicYearRepository.findById(academicYearId)
                        .orElseThrow(() -> new IllegalArgumentException("Annee introuvable."));
                if (!tenantId.equals(year.getEstablishment().getId())) {
                    throw new IllegalStateException("Annee hors de ce locataire.");
                }
            } else {
                year = null;
            }
            Level level = null;
            if (levelId != null) {
                level = levelRepository.findById(levelId)
                        .orElseThrow(() -> new IllegalArgumentException("Niveau introuvable."));
                if (!tenantId.equals(level.getEstablishment().getId())) {
                    throw new IllegalStateException("Niveau hors de ce locataire.");
                }
            }
            Room room = null;
            if (roomId != null) {
                room = roomRepository.findById(roomId)
                        .orElseThrow(() -> new IllegalArgumentException("Salle introuvable."));
                if (!tenantId.equals(room.getEstablishment().getId())) {
                    throw new IllegalStateException("Salle hors de ce locataire.");
                }
            }
            long currentEffectif = 0;
            if (schoolClass.getId() != null) {
                currentEffectif = userRepository.countByEstablishmentIdAndRoles_NameInAndSchoolClass_Id(
                        tenantId, List.of("APPRENANT"), schoolClass.getId());
            }
            if (room != null) {
                List<SchoolClass> sameRoom = schoolClassRepository.findByEstablishmentIdAndRoomId(tenantId, room.getId());
                boolean hasConflict = sameRoom.stream()
                        .filter(existing -> !existing.getId().equals(schoolClass.getId()))
                        .anyMatch(existing -> {
                            if (year == null) {
                                return true;
                            }
                            return existing.getAcademicYear() != null &&
                                    year.getId().equals(existing.getAcademicYear().getId());
                        });
                if (hasConflict && !acceptRoomConflict) {
                    redirectAttributes.addFlashAttribute("error", "Conflit de salle detecte. Confirmez pour continuer.");
                    return "redirect:/tenant/app?tenantId=" + tenantId + "#module-classes";
                }
                Integer roomCapacity = room.getCapacity();
                if (roomCapacity != null && currentEffectif > roomCapacity && !acceptCapacityOverflow) {
                    redirectAttributes.addFlashAttribute("error", "Capacite depassee. Confirmez pour continuer.");
                    return "redirect:/tenant/app?tenantId=" + tenantId + "#module-classes";
                }
            }
            List<SchoolClass> existing = academicYearId != null
                    ? schoolClassRepository.findByEstablishmentIdAndAcademicYearIdOrderByNameAsc(tenantId, academicYearId)
                    : schoolClassRepository.findByEstablishmentIdOrderByNameAsc(tenantId);
            boolean duplicate = existing.stream().anyMatch(c ->
                    !c.getId().equals(classId) &&
                    c.getName() != null &&
                    c.getName().equalsIgnoreCase(trimmedName) &&
                    ((academicYearId == null && c.getAcademicYear() == null) ||
                            (academicYearId != null && c.getAcademicYear() != null &&
                                    academicYearId.equals(c.getAcademicYear().getId())))
            );
            if (duplicate) {
                redirectAttributes.addFlashAttribute("error", "Cette classe existe deja pour cette annee.");
                return "redirect:/tenant/app?tenantId=" + tenantId + "#module-classes";
            }

            if (schoolClass.getId() == null) {
                schoolClass.setEstablishment(current);
            }
            schoolClass.setName(trimmedName);
            schoolClass.setLevel(level);
            schoolClass.setAcademicYear(year);

    private void addLevelTokens(String source, Set<String> names) {
        if (source == null || source.isBlank()) {
            return;
        }
        String[] raw = source.split("[,;\\r\\n]+");
        for (String token : raw) {
            String trimmed = token.trim();
            if (!trimmed.isEmpty()) {
                names.add(trimmed);
            }
        }
    }

    private List<String> buildCandidates(List<User> users, Set<String> roleNames) {
        return users.stream()
                .filter(u -> u.getRoles() != null && u.getRoles().stream()
                        .anyMatch(r -> roleNames.contains(r.getName())))
                .map(this::formatUserName)
                .distinct()
                .toList();
    }

    private String formatUserName(User user) {
        String last = user.getLastName() != null ? user.getLastName().trim() : "";
        String first = user.getFirstName() != null ? user.getFirstName().trim() : "";
        String combined = (last + " " + first).trim();
        if (!combined.isEmpty()) {
            return combined;
        }
        return user.getUsername();
    }

    @PostMapping("/tenant/dashboard/logo")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN')")
    public String updateLogo(
            @RequestParam("logoFile") MultipartFile logoFile,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        Establishment current = est.get();
        String redirect = "redirect:/tenant/app?tenantId=" + current.getId() + "#module-users";

        try {
            if (logoFile == null || logoFile.isEmpty()) {
                redirectAttributes.addFlashAttribute("error", "Veuillez choisir une image.");
                return "redirect:/tenant/app?tenantId=" + est.get().getId() + "#module-users";
            }
            long maxBytes = 300L * 1024L;
            if (logoFile.getSize() > maxBytes) {
                redirectAttributes.addFlashAttribute("error", "Image trop volumineuse (max 300 KB).");
                return "redirect:/tenant/app?tenantId=" + est.get().getId() + "#module-users";
            }
            String contentType = logoFile.getContentType();
            if (contentType == null || !contentType.toLowerCase().startsWith("image/")) {
                redirectAttributes.addFlashAttribute("error", "Format invalide. Choisissez une image PNG/JPG/WebP.");
                return "redirect:/tenant/app?tenantId=" + est.get().getId() + "#module-users";
            }

            BufferedImage original = ImageIO.read(logoFile.getInputStream());
            if (original == null) {
                redirectAttributes.addFlashAttribute("error", "Impossible de lire l'image.");
                return "redirect:/tenant/app?tenantId=" + est.get().getId() + "#module-users";
            }

            int width = original.getWidth();
            int height = original.getHeight();

            int maxDim = 256;
            BufferedImage scaled = original;
            if (Math.max(width, height) > maxDim) {
                double scale = (double) maxDim / (double) Math.max(width, height);
                int newW = Math.max(1, (int) Math.round(width * scale));
                int newH = Math.max(1, (int) Math.round(height * scale));
                BufferedImage out = new BufferedImage(newW, newH, BufferedImage.TYPE_INT_ARGB);
                Graphics2D g = out.createGraphics();
                g.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BICUBIC);
                g.setRenderingHint(RenderingHints.KEY_RENDERING, RenderingHints.VALUE_RENDER_QUALITY);
                g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
                g.drawImage(original, 0, 0, newW, newH, null);
                g.dispose();
                scaled = out;
                width = newW;
                height = newH;
            }

            ByteArrayOutputStream baos = new ByteArrayOutputStream();
            ImageIO.write(scaled, "png", baos);
            byte[] pngBytes = baos.toByteArray();

            Establishment e = est.get();
            e.setLogoUrl("data:image/png;base64," + Base64.getEncoder().encodeToString(pngBytes));
            establishmentRepository.save(e);

            long kb = Math.max(1, pngBytes.length / 1024L);
            redirectAttributes.addFlashAttribute("success", "Logo importe (" + width + "x" + height + ", " + kb + " KB).");
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur import logo : " + ex.getMessage());
        }
        return "redirect:/tenant/dashboard";
    }

                redirectAttributes.addFlashAttribute("error", "Rôle non autorisé pour votre profil.");
                return "redirect:/tenant/app?tenantId=" + est.get().getId() + "#module-users";
            }
            Role r = roleRepository.findByName(role)
                    .orElseThrow(() -> new IllegalArgumentException("Rôle invalide: " + role));

            u.setPassword(passwordEncoder.encode(password));

                u.setPhotoUrl(photoUrl.trim());
            }

            boolean emailAttempted = false;
            String targetEmail = email != null ? email.trim() : "";

                emailAttempted = true;
                String loginUrl = tenantBaseUrl(est.get()) + "/tenant/login";

                emailSent = emailService.sendForTenant(est.get(), targetEmail, subject, body);
            }
            redirectAttributes.addFlashAttribute(
                    "success",
                    emailSent ? "Utilisateur créé et email envoyé." :
                            (emailAttempted ? "Utilisateur créé mais email non envoyé (verifier SMTP)." : "Utilisateur créé.")
            );
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur création utilisateur : " + ex.getMessage());
        }
        return "redirect:/tenant/app?tenantId=" + est.get().getId() + "#module-users";

    @PostMapping("/tenant/app/teachers/import")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN','DIRECTEUR_RH')")
    public String importTeachers(
            @RequestParam(required = false) MultipartFile teachersFile,
            @RequestParam(required = false) String teachersInput,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        Establishment current = est.get();
        Long tenantId = current.getId();
        int created = 0;
        int skipped = 0;
        int missingEmail = 0;
        try {
            Role teacherRole = roleRepository.findByName("ENSEIGNANT")
                    .orElseGet(() -> roleRepository.save(new Role("ENSEIGNANT")));
            Set<String> lines = new LinkedHashSet<>();
            if (teachersFile != null && !teachersFile.isEmpty()) {
                String content = new String(teachersFile.getBytes(), StandardCharsets.UTF_8);
                lines.addAll(List.of(content.split("\\r?\\n")));
            }
            if (teachersInput != null && !teachersInput.isBlank()) {
                lines.addAll(List.of(teachersInput.split("\\r?\\n")));
            }
            for (String line : lines) {
                if (line == null) continue;
                String trimmed = line.trim();
                if (trimmed.isBlank()) continue;
                String lower = trimmed.toLowerCase();
                if (lower.contains("email") && lower.contains("prenom")) {
                    continue;
                }
                String[] parts = trimmed.split("[,;\\t]");
                if (parts.length < 3) {
                    skipped++;
                    continue;
                }
                String firstName = parts[0].trim();
                String lastName = parts[1].trim();
                String email = parts[2].trim();
                String phone = parts.length > 3 ? parts[3].trim() : "";
                String type = parts.length > 4 ? parts[4].trim() : "";
                String className = parts.length > 5 ? parts[5].trim() : "";
                if (email.isBlank()) {
                    missingEmail++;
                    continue;
                }
                String username = email.toLowerCase();
                if (userRepository.findByUsername(username).isPresent()) {
                    skipped++;
                    continue;
                }
                SchoolClass schoolClass = null;
                if (!className.isBlank()) {
                    schoolClass = schoolClassRepository.findByEstablishmentIdAndNameIgnoreCase(tenantId, className)
                            .orElse(null);
                }
                String normalizedType = type != null && !type.isBlank()
                        ? type.trim().toUpperCase()
                        : "TITULAIRE";
                if (!"TITULAIRE".equals(normalizedType) && !"VACATAIRE".equals(normalizedType)) {
                    normalizedType = "TITULAIRE";
                }
                String tempPassword = buildTempTeacherPassword();
                User teacher = new User();
                teacher.setUsername(username);
                teacher.setPassword(passwordEncoder.encode(tempPassword));
                teacher.setFirstName(firstName);
                teacher.setLastName(lastName);
                teacher.setEmail(email);
                teacher.setPhoneMobile(phone.isBlank() ? null : phone);
                teacher.setTeacherType(normalizedType);
                teacher.setSchoolClass(schoolClass);
                teacher.setEnabled(true);
                teacher.setEstablishmentId(tenantId);
                teacher.setArchived(false);
                teacher.getRoles().add(teacherRole);
                userRepository.save(teacher);
                created++;
            }
            redirectAttributes.addFlashAttribute("success",
                    "Import terminé. Créés: " + created + ", ignorés: " + skipped + ", emails manquants: " + missingEmail + ".");
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur import enseignants : " + ex.getMessage());
        }
        return "redirect:/tenant/app?tenantId=" + tenantId + "#module-teachers";
    }

    @PostMapping("/tenant/app/subjects")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN','DIRECTEUR','DIRECTEUR_RH')")
    public String createSubject(
            @RequestParam(required = false) Long subjectId,
            @RequestParam String title,
            @RequestParam(required = false) Integer hoursTd,
            @RequestParam(required = false) Integer hoursTp,
            @RequestParam(required = false) Integer hoursCm,
            @RequestParam(required = false) List<Long> levelIds,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        Long tenantId = est.get().getId();
        try {
            String trimmedTitle = title != null ? title.trim() : "";
            if (trimmedTitle.isBlank()) {
                redirectAttributes.addFlashAttribute("error", "Intitulé requis.");
                return "redirect:/tenant/app?tenantId=" + tenantId + "#module-subjects";
            }
            subjectRepository.findByEstablishmentIdAndTitleIgnoreCase(tenantId, trimmedTitle)
                    .filter(existing -> subjectId == null || !existing.getId().equals(subjectId))
                    .ifPresent(existing -> {
                        throw new IllegalArgumentException("Cette matière existe déjà.");
                    });
            Subject subject = subjectId != null
                    ? subjectRepository.findById(subjectId)
                    .orElseThrow(() -> new IllegalArgumentException("Matière introuvable."))
                    : new Subject();
            if (subject.getEstablishment() != null && !tenantId.equals(subject.getEstablishment().getId())) {
                throw new IllegalStateException("Matière hors de ce locataire.");
            }
            if (subject.getId() == null) {
                subject.setEstablishment(est.get());
            }

                        throw new IllegalStateException("Niveau hors de cet établissement.");
                    }
                    levels.add(level);
                }
            }
            subject.setLevels(levels);
            subjectRepository.save(subject);
            redirectAttributes.addFlashAttribute("success", subjectId == null ? "Matière créée." : "Matière mise à jour.");
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur matière : " + ex.getMessage());

    @PostMapping("/tenant/app/projects/import")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN','DIRECTEUR','DIRECTEUR_RH')")
    public String importProjects(
            @RequestParam(required = false) MultipartFile projectsFile,
            @RequestParam(required = false) String projectsInput,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        Long tenantId = est.get().getId();
        int created = 0;
        int skipped = 0;
        try {
            Set<String> lines = new LinkedHashSet<>();
            if (projectsFile != null && !projectsFile.isEmpty()) {
                String content = new String(projectsFile.getBytes(), StandardCharsets.UTF_8);
                lines.addAll(List.of(content.split("\\r?\\n")));
            }
            if (projectsInput != null && !projectsInput.isBlank()) {
                lines.addAll(List.of(projectsInput.split("\\r?\\n")));
            }
            for (String line : lines) {
                if (line == null) continue;
                String trimmed = line.trim();
                if (trimmed.isBlank()) continue;
                String lower = trimmed.toLowerCase();
                if (lower.contains("titre") && lower.contains("annee")) {
                    continue;
                }
                String[] parts = trimmed.split("[,;\\t]");
                if (parts.length < 1) {
                    skipped++;
                    continue;
                }
                String title = parts[0].trim();
                if (title.isBlank()) {
                    skipped++;
                    continue;
                }
                if (pedagogicalProjectRepository.findByEstablishmentIdAndTitleIgnoreCase(tenantId, title).isPresent()) {
                    skipped++;
                    continue;
                }
                PedagogicalProject project = new PedagogicalProject();
                project.setEstablishment(est.get());
                project.setTitle(title);
                project.setAcademicYear(parts.length > 1 ? parts[1].trim() : null);
                project.setResponsible(parts.length > 2 ? parts[2].trim() : null);
                project.setStatus(parts.length > 3 ? parts[3].trim() : null);
                if (parts.length > 4 && !parts[4].trim().isBlank()) {
                    project.setBudget(new java.math.BigDecimal(parts[4].trim().replace(",", ".")));
                }
                if (parts.length > 5) {
                    StringBuilder desc = new StringBuilder();
                    for (int i = 5; i < parts.length; i++) {
                        if (desc.length() > 0) desc.append(" ");
                        desc.append(parts[i].trim());
                    }
                    project.setDescription(desc.toString());
                }
                pedagogicalProjectRepository.save(project);
                created++;
            }
            redirectAttributes.addFlashAttribute("success", "Import projets terminé. Créés: " + created + ", ignorés: " + skipped + ".");
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur import projets : " + ex.getMessage());
        }
        return "redirect:/tenant/app?tenantId=" + tenantId + "#module-pedagogical_projects";
    }

    @PostMapping("/tenant/app/career-paths")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN','DIRECTEUR','DIRECTEUR_RH')")
    public String saveCareerPath(
            @RequestParam(required = false) Long pathId,
            @RequestParam String name,
            @RequestParam Long levelId,
            @RequestParam(required = false) String description,
            @RequestParam(required = false) List<Long> subjectIds,
            @RequestParam(required = false) List<Long> projectIds,
            @RequestParam(required = false) List<Long> insertionIds,
            @RequestParam(required = false) List<Long> studentIds,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        Long tenantId = est.get().getId();
        try {
            if (name == null || name.isBlank()) {
                throw new IllegalArgumentException("Nom du parcours requis.");
            }
            String trimmedName = name.trim();
            careerPathRepository.findByEstablishmentIdAndNameIgnoreCase(tenantId, trimmedName)
                    .filter(existing -> pathId == null || !existing.getId().equals(pathId))
                    .ifPresent(existing -> {
                        throw new IllegalArgumentException("Ce parcours existe deja.");
                    });
            Level level = levelRepository.findById(levelId)
                    .orElseThrow(() -> new IllegalArgumentException("Niveau introuvable."));
            if (!tenantId.equals(level.getEstablishment().getId())) {
                throw new IllegalStateException("Niveau hors de ce locataire.");
            }
            CareerPath careerPath = pathId == null
                    ? new CareerPath()
                    : careerPathRepository.findById(pathId)
                    .orElseThrow(() -> new IllegalArgumentException("Parcours introuvable."));
            if (careerPath.getEstablishment() != null
                    && !tenantId.equals(careerPath.getEstablishment().getId())) {
                throw new IllegalStateException("Parcours hors de ce locataire.");
            }
            careerPath.setEstablishment(est.get());
            careerPath.setName(trimmedName);
            careerPath.setDescription(description != null && !description.isBlank() ? description.trim() : null);
            careerPath.setLevel(level);
            if (pathId == null) {
                careerPath.setArchived(false);
            }

            List<Subject> subjects = subjectIds == null || subjectIds.isEmpty()
                    ? List.of()
                    : subjectRepository.findAllById(subjectIds);
            for (Subject subject : subjects) {
                if (!tenantId.equals(subject.getEstablishment().getId())) {
                    throw new IllegalStateException("Matiere hors de ce locataire.");
                }
            }
            List<PedagogicalProject> projects = projectIds == null || projectIds.isEmpty()
                    ? List.of()
                    : pedagogicalProjectRepository.findAllById(projectIds);
            for (PedagogicalProject project : projects) {
                if (!tenantId.equals(project.getEstablishment().getId())) {
                    throw new IllegalStateException("Projet hors de ce locataire.");
                }
            }
            List<ProfessionalInsertion> insertions = insertionIds == null || insertionIds.isEmpty()
                    ? List.of()
                    : professionalInsertionRepository.findAllById(insertionIds);
            for (ProfessionalInsertion insertion : insertions) {
                if (!tenantId.equals(insertion.getEstablishment().getId())) {
                    throw new IllegalStateException("Insertion hors de ce locataire.");
                }
            }
            List<User> students = studentIds == null || studentIds.isEmpty()
                    ? List.of()
                    : userRepository.findAllById(studentIds);
            for (User student : students) {
                if (!tenantId.equals(student.getEstablishmentId())) {
                    throw new IllegalStateException("Apprenant hors de ce locataire.");
                }
                boolean isStudent = student.getRoles().stream().anyMatch(r -> "APPRENANT".equals(r.getName()));
                if (!isStudent) {
                    throw new IllegalStateException("Utilisateur non apprenant.");
                }
            }
            careerPath.setSubjects(Set.copyOf(subjects));
            careerPath.setProjects(Set.copyOf(projects));
            careerPath.setInsertions(Set.copyOf(insertions));
            careerPath.setStudents(Set.copyOf(students));
            careerPathRepository.save(careerPath);
            redirectAttributes.addFlashAttribute("success", pathId == null ? "Parcours cree." : "Parcours mis a jour.");
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur parcours : " + ex.getMessage());
        }
        return "redirect:/tenant/app?tenantId=" + tenantId + "#module-career_paths";
    }

            @RequestParam(required = false) MultipartFile photoFile,
            @RequestParam(required = false) MultipartFile[] adminDocuments,
            @RequestParam(required = false) String observation,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        Long tenantId = est.get().getId();
        try {
            User student = userRepository.findById(studentId)
                    .orElseThrow(() -> new IllegalArgumentException("Apprenant introuvable."));
            if (!tenantId.equals(student.getEstablishmentId())) {
                throw new IllegalStateException("Apprenant hors de ce locataire.");
            }
            boolean isStudent = student.getRoles().stream().anyMatch(r -> "APPRENANT".equals(r.getName()));
            if (!isStudent) {
                throw new IllegalStateException("Cet utilisateur n'est pas un apprenant.");
            }
            if (firstName == null || firstName.isBlank() || lastName == null || lastName.isBlank()) {
                throw new IllegalArgumentException("Nom et prenoms sont requis.");
            }
            if (email == null || email.isBlank()) {
                throw new IllegalArgumentException("Email est requis.");
            }
            String username = email.trim().toLowerCase();
            userRepository.findByUsername(username)
                    .filter(existing -> !existing.getId().equals(studentId))
                    .ifPresent(existing -> {
                        throw new IllegalArgumentException("Cet email est déjà utilisé.");
                    });
            SchoolClass schoolClass = null;
            if (schoolClassId != null) {
                schoolClass = schoolClassRepository.findById(schoolClassId)
                        .orElseThrow(() -> new IllegalArgumentException("Classe introuvable."));
                if (!tenantId.equals(schoolClass.getEstablishment().getId())) {
                    throw new IllegalStateException("Classe hors de ce locataire.");
                }
            }
            student.setFirstName(firstName.trim());
            student.setLastName(lastName.trim());
            student.setEmail(email.trim());
            student.setUsername(username);
            student.setPhoneFixed(phoneFixed != null && !phoneFixed.isBlank() ? phoneFixed.trim() : null);
            student.setPhoneMobile(phoneMobile != null && !phoneMobile.isBlank() ? phoneMobile.trim() : null);
            student.setSchoolClass(schoolClass);
            student.setObservation(observation != null && !observation.isBlank() ? observation.trim() : null);
            student.setSubjects(resolveSubjects(subjectIds, tenantId));
            String photoUrl = buildPhotoDataUrl(photoFile);
            if (photoUrl != null) {
                student.setPhotoUrl(photoUrl);
            }
            userRepository.save(student);
            saveStudentDocuments(student, adminDocuments);
            redirectAttributes.addFlashAttribute("success", "Apprenant mis a jour.");
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur apprenant : " + ex.getMessage());
        }
        return "redirect:/tenant/app?tenantId=" + tenantId + "#module-students";
    }

    @PostMapping("/tenant/app/students/import")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN','SECRETAIRE','COMPTABLE')")
    public String importStudents(
            @RequestParam(required = false) MultipartFile studentsFile,
            @RequestParam(required = false) String studentsInput,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        Establishment current = est.get();
        Long tenantId = current.getId();
        int created = 0;
        int skipped = 0;
        int missingEmail = 0;
        try {
            Role studentRole = roleRepository.findByName("APPRENANT")
                    .orElseGet(() -> roleRepository.save(new Role("APPRENANT")));
            Set<String> lines = new LinkedHashSet<>();
            if (studentsFile != null && !studentsFile.isEmpty()) {
                String content = new String(studentsFile.getBytes(), StandardCharsets.UTF_8);
                lines.addAll(List.of(content.split("\\r?\\n")));
            }
            if (studentsInput != null && !studentsInput.isBlank()) {
                lines.addAll(List.of(studentsInput.split("\\r?\\n")));
            }
            for (String line : lines) {
                if (line == null) continue;
                String trimmed = line.trim();
                if (trimmed.isBlank()) continue;
                String lower = trimmed.toLowerCase();
                if (lower.contains("email") && lower.contains("prenom")) {
                    continue;
                }
                String[] parts = trimmed.split("[,;\\t]");
                if (parts.length < 3) {
                    skipped++;
                    continue;
                }
                String firstName = parts[0].trim();
                String lastName = parts[1].trim();
                String email = parts[2].trim();
                String phone = parts.length > 3 ? parts[3].trim() : "";
                String className = parts.length > 4 ? parts[4].trim() : "";
                if (email.isBlank()) {
                    missingEmail++;
                    continue;
                }
                String username = email.toLowerCase();
                if (userRepository.findByUsername(username).isPresent()) {
                    skipped++;
                    continue;
                }
                SchoolClass schoolClass = null;
                if (!className.isBlank()) {
                    schoolClass = schoolClassRepository.findByEstablishmentIdAndNameIgnoreCase(tenantId, className)
                            .orElse(null);
                }
                String tempPassword = buildTempStudentPassword();
                User student = new User();
                student.setUsername(username);
                student.setPassword(passwordEncoder.encode(tempPassword));
                student.setFirstName(firstName);
                student.setLastName(lastName);
                student.setEmail(email);
                student.setPhoneMobile(phone.isBlank() ? null : phone);
                student.setSchoolClass(schoolClass);
                student.setEnabled(true);
                student.setEstablishmentId(tenantId);
                student.setArchived(false);
                student.getRoles().add(studentRole);
                userRepository.save(student);
                created++;
            }
            redirectAttributes.addFlashAttribute("success",
                    "Import terminé. Créés: " + created + ", ignorés: " + skipped + ", emails manquants: " + missingEmail + ".");
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur import apprenants : " + ex.getMessage());
        }
        return "redirect:/tenant/app?tenantId=" + tenantId + "#module-students";
    }

    @PostMapping("/tenant/app/modules/toggle")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN')")

            @RequestParam Long moduleId,
            @RequestParam String status,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        try {
            ModuleDefinition module = moduleDefinitionRepository.findById(moduleId)
                    .orElseThrow(() -> new IllegalArgumentException("Module introuvable"));

        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur module : " + ex.getMessage());

    @PostMapping("/tenant/dashboard/modules/request")
    @PreAuthorize("hasRole('SUPER_ADMIN')")
    public String requestPremiumModule(
            @RequestParam Long moduleId,
            RedirectAttributes redirectAttributes
    ) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";

        try {
            if (est.get().getSubscriptionPlan() != SubscriptionPlan.PREMIUM) {
                redirectAttributes.addFlashAttribute("error", "Option premium inactive. Contactez la direction.");
                return "redirect:/tenant/dashboard";
            }
            ModuleDefinition module = moduleDefinitionRepository.findById(moduleId)
                    .orElseThrow(() -> new IllegalArgumentException("Module introuvable"));
            if (module.getCategory() != ModuleCategory.PREMIUM) {
                redirectAttributes.addFlashAttribute("error", "Ce module est deja inclus dans l'offre de base.");
                return "redirect:/tenant/dashboard";
            }

            Optional<ModuleSubscription> existing = moduleSubscriptionRepository
                    .findByEstablishmentIdAndModuleId(est.get().getId(), moduleId);

            if (existing.isPresent()) {
                ModuleSubscription sub = existing.get();
                if (sub.getStatus() == ModuleStatus.ACTIVE) {
                    redirectAttributes.addFlashAttribute("success", "Module deja actif.");
                    return "redirect:/tenant/dashboard";
                }
                if (sub.getStatus() == ModuleStatus.REQUESTED) {
                    redirectAttributes.addFlashAttribute("success", "Demande deja en cours.");
                    return "redirect:/tenant/dashboard";
                }
                sub.setStatus(ModuleStatus.REQUESTED);
                sub.setRequestedAt(LocalDateTime.now());
                moduleSubscriptionRepository.save(sub);
            } else {
                ModuleSubscription sub = new ModuleSubscription(est.get(), module, ModuleStatus.REQUESTED);
                sub.setRequestedAt(LocalDateTime.now());
                moduleSubscriptionRepository.save(sub);
            }
            redirectAttributes.addFlashAttribute("success", "Demande envoyee a la direction.");
        } catch (Exception ex) {
            redirectAttributes.addFlashAttribute("error", "Erreur demande module : " + ex.getMessage());
        }
        return "redirect:/tenant/dashboard";
    }

    @PostMapping("/tenant/dashboard/complete-setup")
    @PreAuthorize("hasAnyRole('ADMIN','SUPER_ADMIN')")
    public String completeSetup(RedirectAttributes redirectAttributes) {
        Optional<Establishment> est = loadCurrentTenant();
        if (est.isEmpty()) return "error-tenant";
        if (!emailService.isTenantConfigured(est.get())) {
            redirectAttributes.addFlashAttribute("error", "Configurez le SMTP de l'établissement avant de finaliser le paramétrage.");
            return "redirect:/tenant/dashboard";
        }

        Establishment establishment = est.get();
        int created = 0;
        int skipped = 0;
        int missingEmail = 0;
        int emailFailed = 0;
        int emailSent = 0;

        Role userRole = roleRepository.findByName("USER")
                .orElseGet(() -> roleRepository.save(new Role("USER")));

        List<Employee> employees = employeeRepository.findByEstablishmentId(establishment.getId());
        for (Employee employee : employees) {
            String email = employee.getEmail();
            if (email == null || email.isBlank()) {
                missingEmail++;
                continue;
            }
            String username = email.trim().toLowerCase();
            if (userRepository.findByUsername(username).isPresent()) {
                skipped++;
                continue;
            }
            String tempPassword = buildTempPassword(employee);
            User user = new User();
            user.setUsername(username);
            user.setPassword(passwordEncoder.encode(tempPassword));
            user.setEnabled(true);
            user.setEstablishmentId(establishment.getId());
            user.getRoles().add(userRole);
            userRepository.save(user);
            created++;

            String baseUrl = tenantBaseUrl(establishment);
            String loginUrl = baseUrl + "/tenant/login";
            System.out.println("STAFF ACCOUNT: " + username + " / " + tempPassword);
            System.out.println("Login link: " + loginUrl);

            String subject = "OSM - Votre compte personnel";
            String body = buildAccountEmailBody(establishment, username, tempPassword, loginUrl);
            if (emailService.sendForTenant(establishment, email, subject, body)) {
                emailSent++;
            } else {
                emailFailed++;
            }
        }

        if (!establishment.isReady()) {
            establishment.setReady(true);
            establishment.setReadyAt(LocalDateTime.now());
            Authentication auth = SecurityContextHolder.getContext().getAuthentication();
            establishment.setReadyBy(auth != null ? auth.getName() : "SYSTEM");
            establishmentRepository.save(establishment);
        }

        redirectAttributes.addFlashAttribute(
                "success",
                "Parametrage termine. Comptes crees: " + created +
                        ", existants: " + skipped +
                        ", sans email: " + missingEmail +
                        ", emails envoyes: " + emailSent +
                        ", emails en echec: " + emailFailed + "."
        );
        return "redirect:/tenant/dashboard";
    }

    private List<Role> tenantAssignableRoles() {
        Set<String> allowed = allowedRoleNamesForCurrentUser();
        return roleRepository.findAll().stream()
                .filter(r -> allowed.contains(r.getName()))
                .sorted((a, b) -> a.getName().compareToIgnoreCase(b.getName()))
                .toList();
    }

    private Set<String> allowedRoleNamesForCurrentUser() {
        Set<String> currentRoles = currentUserRoleNames();
        Set<String> allowed = new java.util.HashSet<>();
        if (currentRoles.contains("SUPER_ADMIN")) {
            allowed.addAll(Set.of("ADMIN", "DIRECTEUR", "SECRETAIRE", "COMPTABLE", "DIRECTEUR_RH", "ENSEIGNANT", "APPRENANT"));
        }
        if (currentRoles.contains("ADMIN")) {
            allowed.addAll(Set.of("DIRECTEUR", "SECRETAIRE", "COMPTABLE", "DIRECTEUR_RH"));
        }
        if (currentRoles.contains("DIRECTEUR_RH")) {
            allowed.add("ENSEIGNANT");
        }
        if (currentRoles.contains("SECRETAIRE") || currentRoles.contains("COMPTABLE")) {
            allowed.add("APPRENANT");
        }
        return allowed;
    }

    private boolean canManageUser(User user, Set<String> allowedRoles, boolean isSuperAdmin) {
        if (isSuperAdmin) return true;
        Set<String> targetRoles = user.getRoles().stream().map(Role::getName).collect(Collectors.toSet());
        if (targetRoles.contains("ADMIN")) return false;
        for (String role : targetRoles) {
            if (allowedRoles.contains(role)) {
                return true;
            }
        }
        return false;
    }

    private Set<String> currentUserRoleNames() {
        Authentication auth = SecurityContextHolder.getContext().getAuthentication();
        if (auth == null || auth.getAuthorities() == null) return Set.of();
        return auth.getAuthorities().stream()
                .map(a -> a.getAuthority())
                .filter(a -> a.startsWith("ROLE_"))
                .map(a -> a.substring("ROLE_".length()))
                .collect(Collectors.toSet());
    }

    private boolean isSuperAdmin() {
        return currentUserRoleNames().contains("SUPER_ADMIN");
    }

    private Set<Subject> resolveSubjects(List<Long> subjectIds, Long tenantId) {
        if (subjectIds == null || subjectIds.isEmpty()) {
            return new java.util.HashSet<>();
        }
        Set<Subject> subjects = new java.util.HashSet<>();
        for (Long subjectId : subjectIds) {
            if (subjectId == null) continue;
            Subject subject = subjectRepository.findById(subjectId)
                    .orElseThrow(() -> new IllegalArgumentException("Matiere introuvable."));
            if (!tenantId.equals(subject.getEstablishment().getId())) {
                throw new IllegalStateException("Matiere hors de ce locataire.");
            }
            subjects.add(subject);
        }
        return subjects;
    }

    private String buildPhotoDataUrl(MultipartFile photoFile) throws Exception {
        if (photoFile == null || photoFile.isEmpty()) {
            return null;
        }
        String contentType = photoFile.getContentType();
        if (contentType == null || !contentType.startsWith("image/")) {
            throw new IllegalArgumentException("Photo invalide.");
        }
        BufferedImage original = ImageIO.read(photoFile.getInputStream());
        if (original == null) {
            throw new IllegalArgumentException("Photo invalide.");
        }
        int maxSize = 300;
        int width = original.getWidth();
        int height = original.getHeight();
        int maxDim = Math.max(width, height);
        int targetWidth = width;
        int targetHeight = height;
        if (maxDim > maxSize) {
            double scale = (double) maxSize / maxDim;
            targetWidth = (int) Math.round(width * scale);
            targetHeight = (int) Math.round(height * scale);
        }
        int imageType = contentType.contains("png") ? BufferedImage.TYPE_INT_ARGB : BufferedImage.TYPE_INT_RGB;
        BufferedImage resized = new BufferedImage(targetWidth, targetHeight, imageType);
        Graphics2D g = resized.createGraphics();
        g.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
        g.setRenderingHint(RenderingHints.KEY_RENDERING, RenderingHints.VALUE_RENDER_QUALITY);
        g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g.drawImage(original, 0, 0, targetWidth, targetHeight, null);
        g.dispose();
        String format = contentType.contains("png") ? "png" : "jpg";
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        ImageIO.write(resized, format, baos);
        String base64 = Base64.getEncoder().encodeToString(baos.toByteArray());
        return "data:" + contentType + ";base64," + base64;
    }

    private void saveStudentDocuments(User student, MultipartFile[] adminDocuments) throws Exception {
        if (adminDocuments == null || adminDocuments.length == 0) {
            return;
        }
        for (MultipartFile doc : adminDocuments) {
            if (doc == null || doc.isEmpty()) continue;
            String contentType = doc.getContentType();
            if (contentType == null || !(contentType.startsWith("image/") || "application/pdf".equals(contentType))) {
                throw new IllegalArgumentException("Document invalide (PDF ou image).");
            }
            StudentDocument document = new StudentDocument(
                    student,
                    doc.getOriginalFilename(),
                    contentType,
                    doc.getBytes(),
                    LocalDateTime.now()
            );
            studentDocumentRepository.save(document);
        }
    }

    }

    }

    private void applyAcademicYearData(String yearLabel,
                                       String periodType,
                                       String startDate,
                                       String endDate,
                                       String term1Start,
                                       String term1End,
                                       String term2Start,
                                       String term2End,
                                       String term3Start,
                                       String term3End,
                                       AcademicYear year) {
        PeriodType type = "TRIMESTER".equalsIgnoreCase(periodType) ? PeriodType.TRIMESTER : PeriodType.SEMESTER;
        LocalDate start = LocalDate.parse(startDate);
        LocalDate end = LocalDate.parse(endDate);
        LocalDate t1Start = LocalDate.parse(term1Start);
        LocalDate t1End = LocalDate.parse(term1End);
        LocalDate t2Start = LocalDate.parse(term2Start);
        LocalDate t2End = LocalDate.parse(term2End);
        LocalDate t3Start = null;
        LocalDate t3End = null;

        if (type == PeriodType.SEMESTER) {
            boolean hasTrimesterValues = (term3Start != null && !term3Start.isBlank())
                    || (term3End != null && !term3End.isBlank());
            if (hasTrimesterValues) {
                throw new IllegalArgumentException("Organisation invalide : choisissez soit semestres, soit trimestres.");
            }
        }
        if (start.isAfter(end)) {
            throw new IllegalArgumentException("La date de debut doit preceder la date de fin.");
        }
        if (t1Start.isAfter(t1End) || t2Start.isAfter(t2End)) {
            throw new IllegalArgumentException("Dates de periode invalides.");
        }
        if (t1Start.isBefore(start) || t2End.isAfter(end)) {
            throw new IllegalArgumentException("Les periodes doivent etre dans l'annee.");
        }
        if (t1End.isAfter(t2Start)) {
            throw new IllegalArgumentException("La periode 2 doit commencer apres la periode 1.");
        }
        if (type == PeriodType.TRIMESTER) {
            if (term3Start == null || term3Start.isBlank() || term3End == null || term3End.isBlank()) {
                throw new IllegalArgumentException("Les dates du trimestre 3 sont requises.");
            }
            t3Start = LocalDate.parse(term3Start);
            t3End = LocalDate.parse(term3End);
            if (t3Start.isAfter(t3End)) {
                throw new IllegalArgumentException("Dates du trimestre 3 invalides.");
            }
            if (t2End.isAfter(t3Start)) {
                throw new IllegalArgumentException("Le trimestre 3 doit commencer apres le trimestre 2.");
            }
            if (t3End.isAfter(end)) {
                throw new IllegalArgumentException("Le trimestre 3 depasse la fin de l'annee.");
            }
        }

        year.setAcademicYear(yearLabel);
        year.setStartDate(start);
        year.setEndDate(end);
        year.setPeriodType(type);
        year.setTerm1Start(t1Start);
        year.setTerm1End(t1End);
        year.setTerm2Start(t2Start);
        year.setTerm2End(t2End);
        year.setTerm3Start(t3Start);
        year.setTerm3End(t3End);
    }
