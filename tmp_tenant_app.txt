        function buildSubjectMap(selectEl) {
            const map = new Map();
            if (!selectEl) return map;
            Array.from(selectEl.options).forEach((opt) => {
                if (opt.value) {
                    map.set(opt.value, (opt.textContent || '').trim());
                }
            });
            return map;
        }

        function renderSelectedSubjects(listEl, inputsEl, emptyEl, selectedIds, selectEl, readOnly, inputName) {
            if (!listEl || !inputsEl) return;
            listEl.innerHTML = '';
            inputsEl.innerHTML = '';
            const cleaned = (selectedIds || []).map((id) => String(id)).filter(Boolean);
            const uniqueIds = Array.from(new Set(cleaned));
            const resolvedName = inputName || (selectEl && selectEl.dataset ? selectEl.dataset.inputName : null) || 'subjectIds';
            if (selectEl) selectEl.disabled = !!readOnly;
            if (!uniqueIds.length) {
                if (emptyEl) emptyEl.style.display = '';
                return;
            }
            if (emptyEl) emptyEl.style.display = 'none';
            const map = buildSubjectMap(selectEl);
            uniqueIds.forEach((id) => {
                const label = map.get(id) || `#${id}`;
                const item = document.createElement('div');
                item.className = 'selected-item';
                item.dataset.id = id;
                const text = document.createElement('span');
                text.textContent = label;
                item.appendChild(text);
                const hidden = document.createElement('input');
                hidden.type = 'hidden';
                hidden.name = resolvedName;
                hidden.value = id;
                inputsEl.appendChild(hidden);
                if (!readOnly) {
                    const removeBtn = document.createElement('button');
                    removeBtn.type = 'button';
                    removeBtn.setAttribute('aria-label', 'Retirer');
                    removeBtn.textContent = '×';
                    removeBtn.addEventListener('click', () => {
                        item.remove();
                        hidden.remove();
                        if (!listEl.querySelector('.selected-item')) {
                            if (emptyEl) emptyEl.style.display = '';
                        }
                    });
                    item.appendChild(removeBtn);
                }
                listEl.appendChild(item);
            });
        }

        function addSubjectFromSelect(selectEl, listEl, inputsEl, emptyEl, inputName) {
            if (!selectEl || !selectEl.value) return;
            const id = selectEl.value;
            const resolvedName = inputName || (selectEl && selectEl.dataset ? selectEl.dataset.inputName : null) || 'subjectIds';
            if (inputsEl && inputsEl.querySelector(`input[name="${resolvedName}"][value="${id}"]`)) {
                selectEl.value = '';
                return;
            }
            const existing = inputsEl
                ? Array.from(inputsEl.querySelectorAll(`input[name="${resolvedName}"]`)).map((input) => input.value)
                : [];
            renderSelectedSubjects(listEl, inputsEl, emptyEl, [
                ...existing,
                id
            ], selectEl, false, resolvedName);
            selectEl.value = '';
        }

                const rowYear = (row.dataset.year || '').toLowerCase();

                const matchRole = !role || rowRole.includes(role);
                const matchYear = !year || rowYear.includes(year);
