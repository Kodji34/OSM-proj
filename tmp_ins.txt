            if (insertionFile != null && !insertionFile.isEmpty()) {
                String content = new String(insertionFile.getBytes(), StandardCharsets.UTF_8);
                lines.addAll(List.of(content.split("\\r?\\n")));
            }
            if (insertionInput != null && !insertionInput.isBlank()) {

                    continue;

                    continue;
