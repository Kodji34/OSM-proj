                    StringBuilder desc = new StringBuilder();
                    for (int i = 10; i < parts.length; i++) {
                        if (desc.length() > 0) desc.append(" ");
                        desc.append(parts[i].trim());
                    }
                    insertion.setDescription(desc.toString());
                }
                professionalInsertionRepository.save(insertion);
                created++;
            }
            redirectAttributes.addFlashAttribute("success", "Import insertion terminé. Créés: " + created + ", ignorés: " + skipped + ".");

            redirectAttributes.addFlashAttribute("error", "Erreur import insertion : " + ex.getMessage());
        }
        return "redirect:/tenant/app?tenantId=" + tenantId + "#module-insertion_pro";
    }
